{$CLEO .cs}

thread 'switch-char'

1@ = 0

:switch_model

wait 100

if
    0ADC: test_cheat "john"
then
    1@ = 0
    jump @load_model
end

if
    0ADC: test_cheat "bunny"
then
    1@ = 263
    jump @load_model
end

if
    0ADC: test_cheat "sera"
then
    1@ = 264
    jump @load_model
end

if
    0ADC: test_cheat "kat"
then
    1@ = 298
    jump @load_special_model
end

if
    0ADC: test_cheat "layla"
then
    1@ = 300
    jump @load_model
end

if
    0ADC: test_cheat "pony"
then
    1@ = 301
    jump @load_model
end

jump @switch_model









:load_model

    Model.Load(1@)
    038B: load_requested_models

    if
        Model.Available(1@)
    then
        09C7: change_player $PLAYER_CHAR model_to 1@
        0ACA: show_text_box 'MODEL SWITCHED'
    else
        0ACA: show_text_box 'MODEL NOT FOUND'
    end

    wait 1000

    jump @switch_model









:load_special_model
    // Load special models from 290 to 299 in slots 1 to 10

    if
        1@ == 298
    then
        023C: load_special_actor 'CAT' as 9
        038B: load_requested_models

        :wait_cat
        wait 0
        if
            823D: not special_actor 9 loaded
        then
            jump @wait_cat
        end

        09C7: change_player $PLAYER_CHAR model_to 298
        jump @switch_model
    end

    jump @switch_model
